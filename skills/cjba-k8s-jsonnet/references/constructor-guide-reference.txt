// RESOURCE CONSTRUCTOR GUIDE
// This reference returns {} and emits no Kubernetes resources.
// All examples below are comments. Copy a setup and example into your renderer
// when you want to use them; this file is not an environment entry point.
//
// Verified against this repository's pinned dependencies:
//   OneHQ/tanka-lib-k8-onehq: 10a46abb03c8a82ecc9479b036a74d5b79b00f4f
//   jsonnet-libs/k8s-libsonnet: 1.35
// See jsonnetfile.lock.json for the exact dependency revisions.
//
// Contents:
//   A. Shared example settings
//   B. Every resource helper exported by the OneHQ library
//   C. Every supporting helper exported by the OneHQ library
//   D. Native resource examples and custom resources
//   E. Complete native resource/API-object constructor index
//
// k    = Kubernetes constructors from lib/k.libsonnet.
// kone = OneHQ settings adapters from lib/k8-onehq.libsonnet.
// Most kone resource helpers return a spec patch. Start with k.<kind>.new(...),
// add the helper, then add namespace/annotations or unsupported native fields.
// generateSecrets/generateServices return arrays of complete resources;
// crunchydb returns a complete PostgresCluster custom resource.
//
// Paths under vendor/github.com/OneHQ/tanka-lib-k8-onehq/ below refer to the
// upstream helper implementations. Local patches belong in the renderer.

// ============================================================================
// A. SHARED EXAMPLE SETTINGS
// ============================================================================
// Copy this setup once above any examples you use. Values are illustrative.
// BEGIN SETUP
// local k = import 'k.libsonnet';
// local kone = import 'k8-onehq.libsonnet';
// local namespace = 'example';
// local namespaced = { metadata+: { namespace: namespace } };
// local labels = { app: 'example-app', component: 'web' };
// local env = { secretRefs: [], configMaps: [], keyvalue: [['MODE', 'example']] };
// local app = {
//   name: 'app', image: 'example.invalid/application:', tag: 'example-tag',
//   command: ['/bin/sh', '-c', 'echo example'],
//   env: env, resources: { requests: { cpu: '100m', memory: '128Mi' } },
//   volumeMounts: [],
// };
// local workload = {
//   labels: labels, replicas: 1, saName: 'example-account',
//   volumes: [], initC: {}, cont: { app: app },
// };
// local scheduling = { example: { Basic: 'spot', Standard: 'ondemand' } };
// local scheduled = workload + { serviceLevel: 'Basic', resources: scheduling };
// END SETUP
//
// All workload adapters accept containers as an object, not an array:
//   initC: { wait: {...} }, cont: { app: {...}, worker: {...} }
// Container object keys determine std.objectValues order; config.name determines
// each rendered container name. Use distinct names for parallel containers.
// Container image is config.image + config.tag, including ':' or '@' in image.
// registrySecretName is optional on Deployment, StatefulSet, Job and CronJob.
// serviceLevel/resources together enable the scheduling helpers in section C.
// Omit serviceLevel when supplying your own scheduling patches.

// ============================================================================
// B. ALL ONEHQ RESOURCE HELPERS
// ============================================================================

// 1. Deployment -- apps/Deployment.libsonnet
// Signature: kone.createDeploy(values, name)
// Required: labels, replicas, saName, volumes, initC, cont.
// Optional: registrySecretName, serviceLevel/resources,
//           terminationGracePeriodSeconds.
// Sets selectors/Pod labels, replicas, containers, volumes and ServiceAccount.
// Names containing "debug" force one replica and append "-debug" to the
// component label; those labels also change the Deployment selector.
// BEGIN EXAMPLE deployment
// k.apps.v1.deployment.new('example-app')
// + kone.createDeploy(workload, 'example-app')
// + namespaced
// END EXAMPLE

// 2. StatefulSet -- apps/StatefulSet.libsonnet
// Signature: kone.createSS(values, name)
// Required: replicas, serviceName, saName, volumes, initC, cont,
//           volumeClaimTemplates.
// Optional: registrySecretName, serviceLevel/resources.
// Pod labels and selector are {app: name}; values.labels is not used here.
// Create the governing headless Service separately with that selector.
// Native patches can add Pod security settings, startupProbe, named ports,
// PVC metadata and other fields not generated by this pinned helper.
// BEGIN EXAMPLE statefulSet
// k.apps.v1.statefulSet.new('example-stateful')
// + kone.createSS(workload + {
//   serviceName: 'example-stateful-headless',
//   volumeClaimTemplates: [{
//     name: 'data', accessModes: ['ReadWriteOnce'],
//     storageClassName: 'example-storage', storage: '8Gi',
//   }],
// }, 'example-stateful')
// + namespaced
// END EXAMPLE
// The example declares a claim but does not mount it. Add a data volumeMount
// to the selected container when the application should consume the storage.

// 3. Job -- batch/Job.libsonnet
// Signature: kone.generateJob(values, name)
// Required: labels, restartPolicy, volumes, initC, cont.
// Optional: saName, registrySecretName, serviceLevel/resources.
// The helper does not copy backoffLimit, activeDeadlineSeconds, completions,
// parallelism or ttlSecondsAfterFinished; patch these under the Job's spec.
// BEGIN EXAMPLE job
// k.batch.v1.job.new('example-task')
// + kone.generateJob(workload + { restartPolicy: 'Never' }, 'example-task')
// + { spec+: { backoffLimit: 0, activeDeadlineSeconds: 1800 } }
// + namespaced
// END EXAMPLE

// 4. CronJob -- batch/CronJob.libsonnet
// Signature: kone.generateCronJob(values, name)
// Required: schedule, labels, restartPolicy, volumes, initC, cont.
// Optional: registrySecretName, serviceLevel/resources.
// Unlike generateJob, this helper does not copy saName. Patch its nested Pod
// spec when required. Concurrency/history settings belong on CronJob.spec;
// retry/deadline settings belong on spec.jobTemplate.spec.
// BEGIN EXAMPLE cronJob
// k.batch.v1.cronJob.new('example-scheduled-task')
// + kone.generateCronJob(workload + {
//   schedule: '0 * * * *', restartPolicy: 'Never',
// }, 'example-scheduled-task')
// + { spec+: {
//   concurrencyPolicy: 'Forbid', successfulJobsHistoryLimit: 1,
//   jobTemplate+: { spec+: {
//     backoffLimit: 0,
//     template+: { spec+: { serviceAccountName: workload.saName } },
//   } },
// } }
// + namespaced
// END EXAMPLE

// 5. HorizontalPodAutoscaler -- autoscaling/HorizontalPodAutoscaler.libsonnet
// Signature: kone.setHPA(values, name)
// Required: minReplicas, maxReplicas, metrics, behavior.
// name is the target Deployment name. The helper fixes scaleTargetRef to an
// apps/v1 Deployment; override it for another scalable resource.
// BEGIN EXAMPLE horizontalPodAutoscaler
// k.autoscaling.v2.horizontalPodAutoscaler.new('example-autoscaler')
// + kone.setHPA({
//   minReplicas: 1, maxReplicas: 3, behavior: {},
//   metrics: [{ type: 'Resource', resource: {
//     name: 'cpu', target: { type: 'Utilization', averageUtilization: 70 },
//   } }],
// }, 'example-app')
// + namespaced
// END EXAMPLE

// 6. PersistentVolumeClaim -- core/PersistentVolumeClaim.libsonnet
// Signature: kone.createPVC(values)
// Required: accessModes, storageSize, storageClass.
// This is a standalone PVC, separate from StatefulSet volumeClaimTemplates.
// BEGIN EXAMPLE persistentVolumeClaim
// k.core.v1.persistentVolumeClaim.new('example-data')
// + kone.createPVC({
//   accessModes: ['ReadWriteOnce'], storageSize: '8Gi',
//   storageClass: 'example-storage',
// })
// + namespaced
// END EXAMPLE

// 7. Secret array -- core/Secret.libsonnet
// Signature: kone.generateSecrets(values)
// Required: values.secrets = [{name, data, type}, ...].
// Returns an ARRAY of Secrets. data values must already be base64 encoded.
// Add namespace/annotations to each returned Secret, not to the array itself.
// BEGIN EXAMPLE secrets
// [secret + namespaced for secret in kone.generateSecrets({ secrets: [{
//   name: 'secrets-example', type: 'Opaque',
//   data: { EXAMPLE_KEY: std.base64('<EXAMPLE_KEY>') },
// }] })]
// END EXAMPLE
// Use the native Secret constructor for an individually declared Secret:
//   k.core.v1.secret.new(name, data, type='Opaque')
// Registry Secrets use type='kubernetes.io/dockerconfigjson' and a
// '.dockerconfigjson' data key. Base64 is encoding, not encryption.

// 8. Service array -- core/Service.libsonnet
// Signature: kone.generateServices(values)
// Required: values.services = [{name, selector, ports}, ...].
// Returns an ARRAY of Services. It only consumes these three settings; use
// native patches for type, clusterIP, publishNotReadyAddresses, etc.
// BEGIN EXAMPLE services
// [service + namespaced for service in kone.generateServices({ services: [{
//   name: 'example-service', selector: labels,
//   ports: [{ port: 80, targetPort: 8080, protocol: 'TCP' }],
// }] })]
// END EXAMPLE

// 9. Ingress -- networking.k8s.io/Ingress.libsonnet
// Signature: kone.createIngress(hosts, svc_name, port, class, tls, annotations)
// hosts is an ARRAY of host strings. tls is an array or null.
// Generates one '/' Prefix path per host, all targeting the same Service/port.
// Use native rules/path patches for additional routes such as frontend /gql.
// BEGIN EXAMPLE ingress
// k.networking.v1.ingress.new('example-ingress')
// + kone.createIngress(
//   ['example.invalid'], 'example-service', 80, 'example-ingress-class',
//   [{ hosts: ['example.invalid'], secretName: 'example-tls' }],
//   { 'example.invalid/owner': 'example' }
// )
// + namespaced
// END EXAMPLE

// 10. PodDisruptionBudget -- core/PodDisruptionBudget.libsonnet
// Signature: kone.setPDB(values, name)
// Required: selector (a LabelSelector object, such as {matchLabels: labels}).
// Optional: minAvailable OR maxUnavailable; provide one, not both.
// The name argument is currently unused by the helper; native .new sets it.
// BEGIN EXAMPLE podDisruptionBudget
// k.policy.v1.podDisruptionBudget.new('example-budget')
// + kone.setPDB({ selector: { matchLabels: labels }, minAvailable: 1 }, 'example-app')
// + namespaced
// END EXAMPLE

// 11. Crunchy PostgresCluster -- tradetools/crunchy.libsonnet
// Signature: kone.crunchydb(name, config)
// Returns a complete custom resource, including apiVersion/kind/metadata/spec.
// Required config: clusterName, annotations, postgresVersion, users, dataSource,
// instances, patroni, backups, proxy, resources and serviceLevel.
// Each instance needs name, replicas, resources, dataVolumeClaimSpec and
// serviceLevel; affinity and tolerations are optional.
// metadata.name comes from config.clusterName, not the first argument.
// The first name argument is used to match scheduling resource keys.
// BEGIN EXAMPLE postgresCluster
// kone.crunchydb('example-db', {
//   clusterName: 'example-db', annotations: {}, postgresVersion: 17,
//   users: [{ name: 'example-user', databases: ['example_database'] }],
//   dataSource: {}, patroni: {}, backups: {}, proxy: {},
//   resources: scheduling, serviceLevel: 'Basic',
//   instances: [{
//     name: 'primary', replicas: 1, resources: {}, serviceLevel: 'Basic',
//     dataVolumeClaimSpec: {
//       accessModes: ['ReadWriteOnce'], storageClassName: 'example-storage',
//       resources: { requests: { storage: '8Gi' } },
//     },
//   }],
// }) + namespaced
// END EXAMPLE
// Instance, pgBackRest repoHost and pgBouncer scheduling are composed by the
// helper. Operator CRDs and schema validation are provided by the cluster.

// ============================================================================
// C. ALL SUPPORTING ONEHQ HELPERS
// These produce fragments, arrays or transformed resources, not new kinds.
// ============================================================================

// 12. Container array -- core/Container.libsonnet
// Signature: kone.ContainerInception(containerObject)
// Required per container: name, image, tag, command, env, resources, volumeMounts.
// Optional: ports {containerPort, protocol?}, lifecycle, readinessProbe,
// livenessProbe. startupProbe, args, imagePullPolicy, securityContext and
// multiple/named ports need native patches with this pinned implementation.
// std.prune removes null/empty fields from the generated container.
// BEGIN EXAMPLE containers
// kone.ContainerInception({ app: app })
// END EXAMPLE

// 13. Workload container patch -- core/Container.libsonnet
// Signature: kone.generateContainers(init, cont)
// Returns {spec: {template: {spec: {initContainers, containers}}}}.
// Deployment/StatefulSet/Job helpers already invoke it; do not add it twice.
// BEGIN EXAMPLE generateContainers
// kone.generateContainers({}, { app: app })
// END EXAMPLE

// 14. Environment variables -- tradetools/EnvironmentVariables.libsonnet
// Signature: kone.setenv(values)
// Required: secretRefs, configMaps, keyvalue arrays.
// secretRefs entries: [secretName, {ENV_NAME: 'secret-key'}].
// For names beginning 'secrets-', the helper uses ENV_NAME as the Secret key
// and ignores the mapping's value. Other names allow explicit key aliases.
// configMaps also use ENV_NAME as the ConfigMap key, ignoring mapping values.
// keyvalue entries: ['ENV_NAME', 'literal-string'].
// The helper concatenates these sources; it does not remove duplicate env names.
// BEGIN EXAMPLE environment
// kone.setenv({
//   secretRefs: [['example-db-secret', { DBPASSWORD: 'password' }]],
//   configMaps: [['example-config', { MODE: 'MODE' }]],
//   keyvalue: [['FEATURE_ENABLED', 'true']],
// })
// END EXAMPLE

// 15. Volume mounts -- tradetools/VolumeMounts.libsonnet
// Signature: kone.setvolM(volumeMounts)
// Required per item: name, mountPath, readOnly. Optional: subPath.
// BEGIN EXAMPLE volumeMounts
// kone.setvolM([{ name: 'config', mountPath: '/app/settings', readOnly: true }])
// END EXAMPLE

// 16. Volumes -- tradetools/Volumes.libsonnet
// Signature: kone.setvol(volumes)
// Required per item: name, plus claimName OR emptyDir OR secret OR configMap.
// The first recognized source wins in that order. Use native patches for
// projected, downwardAPI, hostPath, CSI or other volume source types.
// BEGIN EXAMPLE volumes
// kone.setvol([
//   { name: 'data', claimName: 'example-data' },
//   { name: 'tmp', emptyDir: {} },
//   { name: 'credentials', secret: { secretName: 'secrets-example' } },
//   { name: 'config', configMap: { name: 'example-config' } },
// ])
// END EXAMPLE

// 17. StatefulSet claim templates -- tradetools/VolumeClaimTemplate.libsonnet
// Signature: kone.generateVolumeClaimTemplates(claims)
// Required per item: name, accessModes, storageClassName, storage.
// Note these names differ from createPVC's storageClass and storageSize.
// BEGIN EXAMPLE volumeClaimTemplates
// kone.generateVolumeClaimTemplates([{
//   name: 'data', accessModes: ['ReadWriteOnce'],
//   storageClassName: 'example-storage', storage: '8Gi',
// }])
// END EXAMPLE

// 18. Scheduling expressions -- core/ServiceLevel.libsonnet
// Signature: kone.fetchServiceLevel(name, resource, serviceLevel)
// Finds resource keys occurring inside name; the first key in sorted object
// order wins. That entry maps serviceLevel to 'spot' or 'ondemand'. Ensure a
// matching key and service level exist; this is substring matching.
// Returns node matchExpressions, not a Kubernetes resource or CPU/memory budget.
// BEGIN EXAMPLE serviceLevel
// kone.fetchServiceLevel('example-app', scheduling, 'Basic')
// END EXAMPLE

// 19. Node affinity -- tradetools/Affinity.libsonnet
// Signature: kone.getAffinity(name, values)
// Required: values.resources and values.serviceLevel.
// Returns a Pod-spec affinity value with required node affinity, selecting the
// preemptible node label according to fetchServiceLevel.
// BEGIN EXAMPLE affinity
// kone.getAffinity('example-app', scheduled)
// END EXAMPLE

// 20. Tolerations -- tradetools/Tolerations.libsonnet
// Signature: kone.getTolerations(name, resource, serviceLevel)
// Uses the same resource-key matching as fetchServiceLevel. Spot returns a
// preemptibleNode=true:NoSchedule toleration; on-demand returns [].
// BEGIN EXAMPLE tolerations
// kone.getTolerations('example-app', scheduling, 'Basic')
// END EXAMPLE

// 21. Ingress host rule -- networking.k8s.io/Ingress.libsonnet
// Signature: kone.set_hosts(host, svc_name, port)
// Returns one rules[] entry with a '/' Prefix path; createIngress invokes it.
// BEGIN EXAMPLE ingressHost
// kone.set_hosts('example.invalid', 'example-service', 80)
// END EXAMPLE

// 22. Clone/patch a Deployment container -- tradetools/ContainerPatch.libsonnet
// Signature: kone.containerPatch(newDeployName, newLabels,
//                              source_containerName, newValues, baseDeployment)
// Rewrites Deployment metadata.name and selectors/Pod labels, then patches the
// matching container. Required newValues: command, resources, env, volumeMounts.
// env and volumeMounts here are already-rendered arrays, not OneHQ settings.
// Optional: lifecycle, readinessProbe, livenessProbe. It keeps the source image,
// renames that container and appends env without deduplicating variable names.
// BEGIN EXAMPLE containerPatch
// kone.containerPatch('example-clone', { app: 'example-clone' }, 'app', {
//   command: ['/bin/sh', '-c', 'echo clone'], resources: {}, env: [], volumeMounts: [],
// }, k.apps.v1.deployment.new('example-app') + kone.createDeploy(workload, 'example-app'))
// + namespaced
// END EXAMPLE

// ============================================================================
// D. NATIVE CONSTRUCTORS AND CUSTOM RESOURCES
// The OneHQ library has no dedicated helpers for the kinds below. Use k directly
// and patch their native spec/data. Section E lists every resource/API-object
// .new constructor present in the pinned native library, beyond these examples.
// ============================================================================

// Namespace is cluster-scoped: do not add metadata.namespace.
// BEGIN EXAMPLE namespace
// k.core.v1.namespace.new(namespace)
// END EXAMPLE

// ConfigMap data values must be strings.
// BEGIN EXAMPLE configMap
// k.core.v1.configMap.new('example-config', {
//   MODE: 'example', 'settings.yaml': std.manifestYamlDoc({ enabled: true }),
// }) + namespaced
// END EXAMPLE

// Individually declared Secret; apply Vault annotations in your renderer.
// BEGIN EXAMPLE secret
// k.core.v1.secret.new('secrets-example', {
//   EXAMPLE_KEY: std.base64('<EXAMPLE_KEY>'),
// }, type='Opaque') + namespaced
// END EXAMPLE

// Standalone Service. All selector entries must match a Pod's labels.
// BEGIN EXAMPLE service
// k.core.v1.service.new('example-service', labels, [
//   { port: 80, targetPort: 8080, protocol: 'TCP' },
// ]) + namespaced
// END EXAMPLE

// A StatefulSet's governing headless Service uses its helper-generated app label.
// BEGIN EXAMPLE headlessService
// k.core.v1.service.new('example-stateful-headless', { app: 'example-stateful' }, [
//   { port: 8080, targetPort: 8080 },
// ]) + { spec+: { clusterIP: 'None', publishNotReadyAddresses: true } } + namespaced
// END EXAMPLE

// ServiceAccount identity and namespaced Job-reading permissions.
// BEGIN EXAMPLE serviceAccount
// k.core.v1.serviceAccount.new('example-account') + namespaced
// END EXAMPLE
// BEGIN EXAMPLE role
// k.rbac.v1.role.new('example-job-reader') + {
//   rules: [{ apiGroups: ['batch'], resources: ['jobs'], verbs: ['get', 'list', 'watch'] }],
// } + namespaced
// END EXAMPLE
// BEGIN EXAMPLE roleBinding
// k.rbac.v1.roleBinding.new('example-job-reader-binding') + {
//   roleRef: { apiGroup: 'rbac.authorization.k8s.io', kind: 'Role', name: 'example-job-reader' },
//   subjects: [{ kind: 'ServiceAccount', name: 'example-account', namespace: namespace }],
// } + namespaced
// END EXAMPLE

// ClusterRole and ClusterRoleBinding have no metadata.namespace. A ServiceAccount
// subject still names its namespace. These are examples, not E2E stack resources.
// BEGIN EXAMPLE clusterRole
// k.rbac.v1.clusterRole.new('example-pod-reader') + {
//   rules: [{ apiGroups: [''], resources: ['pods'], verbs: ['get', 'list'] }],
// }
// END EXAMPLE
// BEGIN EXAMPLE clusterRoleBinding
// k.rbac.v1.clusterRoleBinding.new('example-pod-reader-binding') + {
//   roleRef: { apiGroup: 'rbac.authorization.k8s.io', kind: 'ClusterRole', name: 'example-pod-reader' },
//   subjects: [{ kind: 'ServiceAccount', name: 'example-account', namespace: namespace }],
// }
// END EXAMPLE

// DaemonSet -- one Pod on each matching node; there is no kone.createDaemonSet.
// BEGIN EXAMPLE daemonSet
// k.apps.v1.daemonSet.new('example-agent', containers=kone.ContainerInception({ app: app }))
// + namespaced
// END EXAMPLE

// ReplicaSet -- usually managed by a Deployment.
// BEGIN EXAMPLE replicaSet
// k.apps.v1.replicaSet.new('example-replicaset') + { spec: {
//   replicas: 1, selector: { matchLabels: labels },
//   template: { metadata: { labels: labels }, spec: {
//     containers: kone.ContainerInception({ app: app }),
//   } },
// } } + namespaced
// END EXAMPLE

// Standalone Pod -- container fragments are nested directly under spec.
// BEGIN EXAMPLE pod
// k.core.v1.pod.new('example-pod') + { spec: {
//   restartPolicy: 'Never', containers: kone.ContainerInception({ app: app }),
// } } + namespaced
// END EXAMPLE

// NetworkPolicy -- allow port 8080 from Pods in the same namespace.
// BEGIN EXAMPLE networkPolicy
// k.networking.v1.networkPolicy.new('example-network-policy') + { spec: {
//   podSelector: { matchLabels: labels }, policyTypes: ['Ingress'],
//   ingress: [{ from: [{ podSelector: {} }], ports: [{ protocol: 'TCP', port: 8080 }] }],
// } } + namespaced
// END EXAMPLE

// ResourceQuota and LimitRange -- namespace resource constraints.
// BEGIN EXAMPLE resourceQuota
// k.core.v1.resourceQuota.new('example-quota') + {
//   spec: { hard: { 'requests.cpu': '4', 'requests.memory': '8Gi' } },
// } + namespaced
// END EXAMPLE
// BEGIN EXAMPLE limitRange
// k.core.v1.limitRange.new('example-limits') + { spec: { limits: [{
//   type: 'Container', defaultRequest: { cpu: '100m', memory: '128Mi' },
//   default: { cpu: '1', memory: '512Mi' },
// }] } } + namespaced
// END EXAMPLE

// PersistentVolume -- cluster-scoped storage; PVCs are namespaced claims.
// BEGIN EXAMPLE persistentVolume
// k.core.v1.persistentVolume.new('example-volume') + { spec: {
//   capacity: { storage: '8Gi' }, accessModes: ['ReadWriteOnce'],
//   persistentVolumeReclaimPolicy: 'Retain', storageClassName: 'example-storage',
//   csi: { driver: 'example.invalid/csi', volumeHandle: 'example-volume-id' },
// } }
// END EXAMPLE

// Arbitrary operator kinds use a plain object when no typed constructor exists.
// The API group, scope and spec come from that operator's CRD. For example:
// BEGIN EXAMPLE argoApplication
// {
//   apiVersion: 'argoproj.io/v1alpha1', kind: 'Application',
//   metadata: { name: 'example-application', namespace: 'argocd' },
//   spec: {
//     project: 'default',
//     source: { repoURL: 'https://example.invalid/team/repo.git', targetRevision: 'main', path: 'manifests' },
//     destination: { server: 'https://kubernetes.default.svc', namespace: namespace },
//   },
// }
// END EXAMPLE
// The same plain-object pattern applies to ApplicationSet, AppProject and other
// installed CRDs. crunchydb is the OneHQ-specific PostgresCluster constructor.

// Generic List -- native library extension for grouping resources in one object.
// BEGIN EXAMPLE list
// k.core.v1.list.new([
//   k.core.v1.namespace.new(namespace),
//   k.core.v1.configMap.new('example-config', { MODE: 'example' }) + namespaced,
// ])
// END EXAMPLE
